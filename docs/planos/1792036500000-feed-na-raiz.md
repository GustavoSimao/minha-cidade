```tsx
import { redirect } from "next/navigation";

export const instant = false;

interface FeedRedirectProps {
  searchParams: Promise<Record<string, string | string[] | undefined>>;
}

/** `/feed` (e `/feed?...`) redireciona para `/` preservando os filtros. */
export default async function FeedRedirectPage({ searchParams }: FeedRedirectProps) {
  const params = await searchParams;
  const query = new URLSearchParams();

  for (const [key, value] of Object.entries(params)) {
    if (typeof value === "string") {
      query.set(key, value);
    }
  }

  const search = query.toString();
  redirect(search === "" ? "/" : `/?${search}`);
}
```

Decisão: **`/` é a rota canônica**; `/feed` existe só como redirect (SEO e e2e com URL
limpa). Alternativa rejeitada: alias renderando o feed nas duas rotas (duas shells PPR,
`revalidatePath` em dobrado, dois lugares para dar drift).
